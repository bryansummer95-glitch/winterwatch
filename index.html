function calculateDeparture(freezeTime, length) { if (!freezeTime) { return { time: "Not calculable — no freeze forecast", valid: true, reason: "No freezing hour forecast. No departure time can be calculated." }; } // Finish spreading at least one hour before freezing. // Travel time: one hour per 20 km. const required = new Date( new Date(freezeTime).getTime() - (1 + length / 20) * 3600000 ); const hour = Number( required.toLocaleString("en-IE", { timeZone: "Europe/Dublin", hour: "2-digit", hour12: false }) ); const minute = Number( required.toLocaleString("en-IE", { timeZone: "Europe/Dublin", minute: "2-digit" }) ); const minutesOfDay = hour * 60 + minute; // Permitted departure window: 17:00 to 05:00. const insideWindow = minutesOfDay >= 17 * 60 || minutesOfDay <= 5 * 60; return { time: timeString(required), valid: insideWindow, reason: insideWindow ? "" : "Required departure is outside the permitted 17:00–05:00 window." }; } function calculateRoute(route, weather, depotMap) { const h = weather.hourly; if ( !h || !h.soil_temperature_0cm || !h.dew_point_2m || !h.precipitation || !h.time ) { throw new Error("Missing forecast data for " + route.id); } const soil = h.soil_temperature_0cm.slice(0, 18); const dew = h.dew_point_2m.slice(0, 18); const rain = h.precipitation.slice(0, 18); const times = h.time.slice(0, 18); const valid = soil .map((v, i) => ({ v, i })) .filter(x => Number.isFinite(x.v)); if (!valid.length) { throw new Error("No valid ground temperature for " + route.id); } const minimum = Math.min(...valid.map(x => x.v)); let wet = false; let freezeTime = null; for (let i = 0; i < soil.length; i++) { if (!Number.isFinite(soil[i])) continue; if (soil[i] <= 0 && freezeTime === null) { freezeTime = times[i]; } if ( (Number.isFinite(rain[i]) && rain[i] > 0) || ( Number.isFinite(dew[i]) && Math.abs(soil[i] - dew[i]) <= 1 ) ) { wet = true; } } const limit = route.priority === 1 ? 1.0 : route.priority === 2 ? 0.5 : -1.0; const depot = depotMap.get(norm(route.depot)); let decision = "DON'T TREAT"; let reason = ""; if (minimum > limit) { reason = `Forecast minimum ${minimum.toFixed(1)}°C is above the ` + `${limit.toFixed(1)}°C Priority ${route.priority} threshold.`; } else if (!wet) { reason = `Forecast minimum ${minimum.toFixed(1)}°C meets the temperature ` + "threshold, but the road is not forecast to be wet."; } else if (!depot) { reason = "Depot information unavailable. Engineer must verify resources."; } else if ( route.priority === 3 && depot.capacity && depot.stock < depot.capacity * 0.40 ) { reason = `Priority 3 requires at least 40% depot stock. ` + `Current stock: ${(depot.stock / depot.capacity * 100).toFixed(1)}%.`; } else if (route.salt <= 0) { reason = "Salt requirement unavailable."; } else { // Weather conditions support treatment. // Resource and departure checks happen in allocateResources(). decision = "TREAT"; reason = `Minimum ground temperature ${minimum.toFixed(1)}°C meets the ` + `${limit.toFixed(1)}°C threshold and the road may be wet.`; } return { ...route, minimum, wet, freezeTime, depot, decision, reason, departure: "—" }; }
